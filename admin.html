// Google Apps Script (Code.gs)
function getOrCreateSheet() {
  try {
    var active = SpreadsheetApp.getActiveSpreadsheet();
    if (active) return active.getActiveSheet();
  } catch(e) {}
  var props = PropertiesService.getScriptProperties();
  var sheetId = props.getProperty("ABLE_SHEET_ID");
  if (sheetId) {
    try {
      var ss = SpreadsheetApp.openById(sheetId);
      if (ss) return ss.getActiveSheet();
    } catch(err) {}
  }
  var newSS = SpreadsheetApp.create("ABLE Pitch Registrations - November 2026");
  var sheet = newSS.getActiveSheet();
  sheet.appendRow(["Ticket ID", "Full Name", "Email Address", "Grade Level", "Competing Status", "Attendance Venue", "Country", "Timestamp"]);
  props.setProperty("ABLE_SHEET_ID", newSS.getId());
  return sheet;
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = getOrCreateSheet();
    var params = e.parameter || {};
    if (e.postData && e.postData.contents) {
      try { 
        var parsed = JSON.parse(e.postData.contents);
        for (var k in parsed) { params[k] = parsed[k]; }
      } catch(err) {}
    }
    var ticketId = params.id || params.ticket_id || ("ABLE-PITCH-" + Math.floor(1000 + Math.random() * 9000));
    var name = params.name || "";
    var email = params.email || "";
    var grade = params.gradeLevel || params.grade_level || "";
    var status = params.competingStatus || params.competing_status || "Yes";
    var venue = params.venueAttendance || params.venue_attendance || "Online Zoom";
    var country = params.country || "Global";
    var timestamp = new Date().toLocaleString();
    sheet.appendRow([ticketId, name, email, grade, status, venue, country, timestamp]);

    if (email) {
      var subject = "Registration Confirmed: ABLE Business Startup Pitch Challenge 2026";
      var body = "Dear " + name + ",\n\n" +
                 "Congratulations! Your registration for the ABLE Business Startup Pitch Challenge (November 2026) has been successfully confirmed.\n\n" +
                 "Your Official Entry Details:\n" +
                 "• Ticket ID: " + ticketId + "\n" +
                 "• Participant Name: " + name + "\n" +
                 "• Grade Level: " + grade + "\n" +
                 "• Selected Venue: " + venue + "\n" +
                 "• Grand Prize: $250 1st Place Cash Award\n\n" +
                 "Next Steps:\n" +
                 "We will email your 1-Week Prep Schedule, expert mentor coaching sessions, and Colorado Springs venue / Zoom livestream credentials shortly.\n\n" +
                 "Sincerely,\nABLE Academy of Business & Leadership Empowerment\nalfaizablebusines@gmail.com";
      GmailApp.sendEmail(email, subject, body, { name: "ABLE Youth Initiative", replyTo: "alfaizablebusines@gmail.com" });
    }
    return ContentService.createTextOutput(JSON.stringify({ "result": "success", "id": ticketId })).setMimeType(ContentService.MimeType.JSON);
  } catch(e) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": e.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  try {
    var sheet = getOrCreateSheet();
    var rows = sheet.getDataRange().getValues();
    var list = [];
    for (var i = 1; i < rows.length; i++) {
      var r = rows[i];
      if (r[0]) {
        list.push({ id: String(r[0]), name: String(r[1]), email: String(r[2]), gradeLevel: String(r[3]), competingStatus: String(r[4]), venueAttendance: String(r[5]), country: String(r[6]), timestamp: String(r[7]) });
      }
    }
    return ContentService.createTextOutput(JSON.stringify(list)).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
}
